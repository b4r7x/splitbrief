### Part B: HTML renderer

Create `src/engine/export/html-renderer.ts`:

```typescript
import type { ExportData } from './types.js';

export function renderSessionHtml(data: ExportData): string {
  return `<!DOCTYPE html>
<html lang="en">
<head>
<meta charset="utf-8">
<meta name="viewport" content="width=device-width, initial-scale=1">
<title>diptych — ${escapeHtml(data.feature)}</title>
<style>${CSS}</style>
</head>
<body>
<main>
${renderHeader(data)}
${renderHeroSavings(data)}
${renderMetadata(data)}
${renderTaskSummary(data)}
${data.evidence ? renderEvidence(data.evidence) : ''}
${data.drift ? renderDrift(data.drift) : ''}
${data.briefQuality ? renderBriefQuality(data.briefQuality) : ''}
${renderPhaseTiming(data)}
${renderFooter(data)}
</main>
</body>
</html>`;
}
```

**CSS design:** Dark theme matching TUI visual language. Background `#0a0a0a`, text `#e0e0e0`, accent `#00bcd4` (teal). Sections use background stepping (`#141414` → `#1e1e1e`). Left accent borders on sections. Monospace font. Max-width `800px`, centered.

**Sections to render:**

1. **Header** — `diptych` logo text + feature name + completion date.
2. **Hero savings** — Large text: `$X.XX actual vs $Y.YY all-planner — ZZ% saved`. Green accent color. If implementer is local/unpriced, show `local` instead of dollar amount.
3. **Metadata** — Table: planner tool/model, implementer tool/model, mode, total time.
4. **Task summary** — `N tasks: M completed locally, K escalated, J failed, L skipped`. Colored badges.
5. **Evidence** (optional) — `N/M tasks with validation evidence`. Link note to `evidence.json`.
6. **Drift** (optional) — Score badge (green/yellow/red), error/warning counts.
7. **Brief quality** (optional) — Score badge, pass/fail.
8. **Phase timing** — Horizontal bars showing time per phase (planning, implementing, reviewing). Proportional width.
9. **Footer** — `Generated by diptych · {date}`. Muted.

**Helper:**

```typescript
function escapeHtml(s: string): string {
  return s.replace(/&/g, '&amp;').replace(/</g, '&lt;').replace(/>/g, '&gt;').replace(/"/g, '&quot;');
}
```

**Cost formatting:** Use the same logic as `src/features/summary/components/hero-savings.tsx` for savings display. If `costBreakdown` is present, use `savingsPercentage`, `totalActualCost`, `hypotheticalCost`. If absent, omit the hero savings section.

### CSS constant

Define `const CSS` as a template literal string with all styles. Keep it under 150 lines. Mobile-responsive via `max-width` and percentage widths. No media queries needed for a report page.

```typescript
// src/engine/export/html-renderer.test.ts
import { describe, it, expect } from 'vitest';
import { renderSessionHtml } from './html-renderer.js';
import type { ExportData } from './types.js';

function makeExportData(overrides?: Partial<ExportData>): ExportData {
  return {
    sessionId: '2026-05-04-test-feature',
    feature: 'add user auth',
    completedAt: '2026-05-04T12:00:00Z',
    summary: {
      feature: 'add user auth',
      totalTasks: 5,
      completedByLocal: 4,
      escalatedToPlanner: 1,
      skipped: 0,
      failed: 0,
      totalTime: 120_000,
      tokenUsage: { plannerInput: 50000, plannerOutput: 10000, implementerInput: 20000, implementerOutput: 5000 },
      costBreakdown: {
        hypotheticalCost: 1.20,
        actualPlannerCost: 0.15,
        actualImplementerCost: 0.02,
        totalActualCost: 0.17,
        savingsAmount: 1.03,
        savingsPercentage: 85.8,
        localCompletionRate: 0.8,
      },
      plannerTool: 'claude-code',
      plannerModel: 'claude-opus-4-5',
      implementerTool: 'ollama',
      implementerModel: 'qwen2.5-coder:7b',
      phaseTimings: { researching: 30000, specifying: 20000, planning: 15000, implementing: 50000, 'final-review': 5000 },
    },
    ...overrides,
  };
}

describe('renderSessionHtml', () => {
  it('produces HTML with DOCTYPE', () => {
    const html = renderSessionHtml(makeExportData());
    expect(html).toMatch(/^<!DOCTYPE html>/);
  });

  it('includes feature name in title', () => {
    const html = renderSessionHtml(makeExportData());
    expect(html).toContain('<title>diptych — add user auth</title>');
  });

  it('includes hero savings when cost data exists', () => {
    const html = renderSessionHtml(makeExportData());
    expect(html).toContain('85.8%');
    expect(html).toContain('$0.17');
  });

  it('omits hero savings when no cost breakdown', () => {
    const data = makeExportData();
    delete (data.summary as Record<string, unknown>).costBreakdown;
    const html = renderSessionHtml(data);
    expect(html).not.toContain('saved');
  });

  it('includes evidence section when evidence data present', () => {
    const html = renderSessionHtml(makeExportData({
      evidence: { totalTasks: 5, tasksWithValidationEvidence: 4, escalatedTasks: 1, failedTasks: 0 },
    }));
    expect(html).toContain('4/5');
  });

  it('omits evidence section when no evidence data', () => {
    const html = renderSessionHtml(makeExportData());
    expect(html).not.toContain('Evidence');
  });

  it('escapes HTML in feature name', () => {
    const html = renderSessionHtml(makeExportData({ feature: 'fix <script>alert("xss")</script>' }));
    expect(html).not.toContain('<script>');
    expect(html).toContain('&lt;script&gt;');
  });

  it('includes inline CSS', () => {
    const html = renderSessionHtml(makeExportData());
    expect(html).toContain('<style>');
    expect(html).not.toContain('<link');
  });

  it('includes phase timing bars', () => {
    const html = renderSessionHtml(makeExportData());
    expect(html).toContain('researching');
    expect(html).toContain('implementing');
  });
});
```
